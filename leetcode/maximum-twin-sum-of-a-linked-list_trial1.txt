# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def pairSum(self, head: Optional[ListNode]) -> int:
        slow = head
        fast = head.next

        while fast and fast.next:
            fast=fast.next.next
            slow=slow.next
        
        left = head
        right = slow.next

        # Detach and rotate right array
        slow.next = None
        prev = None
        cur = right

        while cur:
            nextNode = cur.next
            cur.next = prev
            prev = cur
            cur=nextNode
        
        # compare prev and left
        maxSum = 0
        while left and prev:
            maxSum=max(left.val+prev.val,maxSum)
            left=left.next
            prev=prev.next
        
        return maxSum
